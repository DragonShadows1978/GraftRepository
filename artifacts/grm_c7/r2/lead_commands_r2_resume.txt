if [[ "${1:-}" == "--check" ]]; then exit 0; fi
if [[ $# -ne 0 ]]; then echo 'Unexpected argument' >&2; exit 2; fi
run_cell() {
    local cell_id="$1"
    local cell_dir="artifacts/grm_c7/r2/fix4_attempt_1/cells/$cell_id"
    if [[ -e "$cell_dir" ]]; then
        # The controller owns skipping/validation: this entrypoint never
        # overwrites or retries a started attempt, including a failed one.
        python - "$cell_dir" <<'CHECK_CELL'
import sys
from pathlib import Path
from scripts.grm_c7_common import read, validate_checkpoint
from scripts.grm_c7_run import binding
p=Path(sys.argv[1])
c=read(p/'controller.json')
assert c['status']=='COMPLETE', 'RED: started cell cannot be retried'
assert c['binding']==binding('A'), 'RED: completed cell binding changed'
validate_checkpoint(p/'checkpoint', c['cell']['stop']+1, binding('A'))
CHECK_CELL
    else
        timeout --signal=INT 590 python scripts/grm_c7_run.py --cell "$cell_id"
    fi
}

# Prior art: C7/C2 foreground leased cells (GRM contributors, 2026).
# r2 fresh 2 GPU-h cap; cumulative C7 cap 3.9 GPU-h. B stays NON_FIT.
# CPU gates were run against the SHA-bound amendment; verify their receipt.
CUDA_VISIBLE_DEVICES='' python scripts/grm_c7_run.py --dry-run
CUDA_VISIBLE_DEVICES='' python - <<'CHECK_CPU'
from scripts.grm_c7_common import OUT, read, verify
r=verify()
b=read(OUT/'r2/cpu_registered.json'); m=read(OUT/'r2/cpu_mutations.json')
assert b['returncode']==0 and b['executed_inputs']==r['effective_inputs']
assert b['amendment_sha256']==m['amendment_sha256']==r['amendment_r2_sha256']
assert m['pass'] and m['live_sources_unchanged']
CHECK_CPU
run_cell() {
    local cell_id="$1"
    local cell_dir="artifacts/grm_c7/r2/cells/$cell_id"
    if [ -f "$cell_dir/controller.json" ]; then
        python - "$cell_dir/controller.json" <<'CHECK_COMPLETE'
import json, sys
from scripts.grm_c7_run import binding
row=json.load(open(sys.argv[1]))
if row['status']!='COMPLETE' or row['binding']!=binding('A'):
    raise SystemExit('RED: incomplete/stale started cell; never retry')
print('Already complete:', row['cell']['id'])
CHECK_COMPLETE
    elif [ -e "$cell_dir" ]; then
        echo "RED: orphaned started cell $cell_id; no retry" >&2
        return 1
    else
        # Owned worker 280s; lease 285s; outer 590s; foreground cooldown 30s.
        # Never clear shared locks or owner files; no retries after RED.
        timeout --signal=INT 590 python scripts/grm_c7_run.py --cell "$cell_id"
    fi
}

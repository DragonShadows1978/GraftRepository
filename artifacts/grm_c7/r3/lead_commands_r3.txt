# Prior art: C7/C2 foreground leased cells (GRM contributors, 2026).
# Fresh r3: FIX3/4/5 are source-pinned ON; FIX7 absent. 7920s total cap.
# This is a LEAD GPU command. --check-only runs CPU preflight/receipt checks.
CUDA_VISIBLE_DEVICES='' python -m scripts.grm_c7_register_r3 --preflight
CUDA_VISIBLE_DEVICES='' python -m scripts.grm_c7_register_r3 --ready

if [[ "${1:-}" == "--check-only" ]]; then exit 0; fi
if [[ $# -ne 0 ]]; then echo 'Unsupported argument' >&2; exit 2; fi
run_cell() {
    local cell_id="$1"
    local status
    status="$(CUDA_VISIBLE_DEVICES='' python - "$cell_id" <<'CHECK_COMPLETE'
import sys
from scripts.grm_c7_register_r3 import resume_check
print(resume_check(sys.argv[1])['status'])
CHECK_COMPLETE
)"
    if [[ "$status" == "COMPLETE" ]]; then
        echo "Already complete and checkpoint verified: $cell_id"
    elif [[ "$status" == "UNSTARTED" ]]; then
        CUDA_VISIBLE_DEVICES='' python -m scripts.grm_c7_register_r3 --preflight
        # Existing owned worker 280s; lease 285s; foreground cooldown 30s.
        # No retry, lock clearing, process-name kill or background job.
        timeout --signal=INT 590 python scripts/grm_c7_run.py --cell "$cell_id"
    else
        echo "RED: unexpected resume state $status" >&2
        return 1
    fi
}

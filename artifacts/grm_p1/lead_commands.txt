import json, sys
sys.path.insert(0, '.')
from scripts import grm_lt1 as lt
from scripts.grm_chat import RECAP_QUESTION
from scripts.grm_p1_smoke import RECALLS
session = sys.argv[1]
rows = [json.loads(l) for l in
        open(f'{session}/session_ledger.jsonl') if l.strip()]
turns = {r['turn']: r for r in rows if r.get('schema') == 'grm.chat_turn.v1'}
restarts = [r for r in rows if r.get('kind') == 'restart']
# A turn that raised has answer=None and an `error` field; it is evidence,
# not an excuse to stop scoring.
failed = [{'turn': t['turn'], 'user': t['user'], 'error': t['error']}
          for t in turns.values() if t.get('error')]
out = []
for spec in RECALLS:
    row = turns.get(spec['turn'])
    if row is None:
        out.append(dict(spec, answer=None, mounted=None, correct=False,
                        category='MISSING_ROW'))
        continue
    if row.get('error'):
        out.append(dict(spec, answer=None, mounted=[], correct=False,
                        category='TURN_FAILED', error=row['error']))
        continue
    s = lt.score(row['answer'], spec['expected'])
    out.append(dict(spec, answer=row['answer'], mounted=row['mounted_ids'],
                    correct=bool(s['exact_correct']), category=s['category']))
fresh = [r for r in out if r['klass'] == 'fresh']
# The recap is found by its QUESTION, never by a fixed turn number: it is
# the last turn only when nothing before it failed, and r2 proved that is
# not guaranteed.
recap = next((t for t in turns.values() if t['user'] == RECAP_QUESTION), None)
print(json.dumps({
    'session': session,
    'turns': len(turns),
    'failed_turns': failed,
    'receipts_ok': all(t['route_receipt'] is not None
                       and t['route_receipt']['schema'] == 'grm.route_receipt.v1'
                       for t in turns.values() if not t.get('error')),
    'restart_retained': all(r['retained'] for r in restarts),
    'fresh_correct': sum(r['correct'] for r in fresh), 'fresh_of': len(fresh),
    'GATE_fresh_ge_3_of_4': sum(r['correct'] for r in fresh) >= 3,
    'corrections_reported': [r for r in out if r['klass'] == 'correction'],
    'alias_reported': [r for r in out if r['klass'] == 'alias'],
    'recap_turn': recap['turn'] if recap else None,
    'recap_answer': (recap or {}).get('answer'),
    'recap_error': (recap or {}).get('error'),
    'rows': out,
}, indent=2))
PY

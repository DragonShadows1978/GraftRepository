#!/usr/bin/env bash
# GRM-A1 — LEAD GPU commands for the registered alias fold-merge contrast.
#
# REVISION 3 (2026-09-11). The previous revision told you to take ONE 2400 s
# lease. That is above the 590 s house cap and died before a single row ran.
# This revision drives a LOOP of <=590 s leases with --resume, which is what
# the resumable create-only receipts were built for.
#
# Registration (written BEFORE any GPU run):
#   artifacts/grm_a1/gpu_contrast_registration.json
#   sha256 329d5865d437598f41fe9569fe8c06a4c2da64544476cf3ee73eb5e3efa9d8cd
# Amendment 1 — worker becomes a pinned input; LT1 RED retracted.
# Amendment 2 — lease shape, declared frame/flags-source/preconditions, one
#   shared flags+environment path (the three 2026-09-11 lead-run stops).
#   Both chain to the registration sha above; .sha256 sidecars alongside.
#
# Registered prediction, UNCHANGED by either amendment:
#   aliases >= 5/8 exact from a SINGLE digest mount; 0 controls broken.
# Budget ceiling, UNCHANGED: 0.5 GPU-h TOTAL across invocations. Splitting
# the lease does not change the budget.
#
# GPU CONTENTION: the worker leases through the SAME gpu_lease helper the
# C7/R1 workers import. The operator has absolute right of way. NEVER kill or
# signal a process this did not start.

set -euo pipefail
cd /mnt/ForgeRealm/wt/grm-a1

# ---------------------------------------------------------------------------
# 0. CPU receipts (all green in the seat's run; re-runnable, no GPU)
# ---------------------------------------------------------------------------
CUDA_VISIBLE_DEVICES='' python3 -m pytest -q \
    tests/test_grm_a1_alias_fold.py tests/test_grm_a1_gpu_contrast.py

CUDA_VISIBLE_DEVICES='' python3 scripts/grm_a1_off_parity.py \
    /mnt/ForgeRealm/wt/grm-merge --json artifacts/grm_a1/off_parity.json

CUDA_VISIBLE_DEVICES='' python3 scripts/grm_a1_width_receipts.py \
    --json artifacts/grm_a1/width_receipts.json

# All 23 rows on the CPU doubles, through the SAME loader helpers the GPU arm
# uses (resolve_flags / build_environment / load_runtime_frame). Only the
# model and tokenizer are stubbed, so a flags-shape or frame error fails here.
CUDA_VISIBLE_DEVICES='' python3 scripts/grm_a1_gpu_contrast.py \
    --fake --out artifacts/grm_a1/gpu_fake

# ---------------------------------------------------------------------------
# 1. PRECONDITION CHECK — run this FIRST. No lease, no load, no card.
# ---------------------------------------------------------------------------
# Reports every declared input and precondition, and names the missing ones.
# Exit 0 = ready; exit 2 = something is missing (it prints which).
CUDA_VISIBLE_DEVICES='' python3 scripts/grm_a1_gpu_contrast.py \
    --dry-run --out artifacts/grm_a1/gpu

# Checked by --dry-run:
#   input        runtime_frame       artifacts/grm_det1/run_20260831T160525Z_2/
#                                    runtime_frame_28b3196f8fb04a41.json  (sha-pinned)
#   input        c7_r3_registration  /mnt/ForgeRealm/wt/grm-c7/artifacts/grm_c7/
#                                    r3/registration.json  (sha-pinned; arms.A.flags
#                                    is THE flags source for these checkpoints)
#   precondition native_runtime      cpp/build/libgrm_runtime.so  (build artifact,
#                                    not sha-pinned; cpp/ unchanged by A1)
#   precondition model_snapshot      ~/.cache/huggingface/hub/models--openai--
#                                    gpt-oss-20b/snapshots/6cee5e81ee83917806bb...
#   checkpoint   (all 15 pinned C7 r3 cells)

# ---------------------------------------------------------------------------
# 2. The GPU contrast (LEAD ONLY) — a LOOP of <=590 s leases
# ---------------------------------------------------------------------------
# Each invocation: verifies drift + preconditions (no lease yet), takes ONE
# 560 s lease, runs as many rows as fit, writes create-only receipts, and
# exits. An expired lease is caught and reported as a PAUSE, not a crash.
#
# Exit codes: 0 = prediction met · 1 = prediction NOT met (a real RED result)
#             2 = precondition failed (drift/chain/pin/missing)
#             3 = unexpected exception · 4 = rows still pending, re-run
#
# The loop stops on 0 or 1 (a verdict), retries on 4, and aborts on 2 or 3.

for attempt in $(seq 1 12); do
    set +e
    python3 scripts/grm_a1_gpu_contrast.py \
        --registration artifacts/grm_a1/gpu_contrast_registration.json \
        --out artifacts/grm_a1/gpu \
        --lease-seconds 560 \
        --resume
    rc=$?
    set -e
    case "$rc" in
        0|1) echo "verdict reached (exit $rc) after $attempt lease(s)"; break ;;
        4)   echo "lease $attempt done, rows pending — continuing"; sleep 30 ;;
        *)   echo "ABORT: exit $rc" >&2; exit "$rc" ;;
    esac
done

# ---------------------------------------------------------------------------
# 3. Summary
# ---------------------------------------------------------------------------
# The worker writes artifacts/grm_a1/gpu/{summary.json,probes.jsonl} itself.
CUDA_VISIBLE_DEVICES='' python3 - <<'SUMMARY'
import json, pathlib
summary = json.loads(
    pathlib.Path('artifacts/grm_a1/gpu/summary.json').read_text())
print(json.dumps(summary, indent=1, sort_keys=True))
print('aliases exact %s/%s  single-mount %s/%s  controls broken %s/%s  pending %s'
      % (summary['aliases_exact'], summary['scored_rows'],
         summary['aliases_single_mount'], summary['scored_rows'],
         summary['controls_broken'], summary['controls'],
         summary.get('pending')))
if summary.get('pending'):
    print('INCOMPLETE — re-run the lease loop above')
else:
    print('PREDICTION MET' if summary['prediction_met']
          else 'PREDICTION NOT MET (RED)')
SUMMARY

# ---------------------------------------------------------------------------
# 4. Optional: LT1 under its REGISTERED margin_first rule
# ---------------------------------------------------------------------------
# LT1 is NOT part of the GPU contrast (its rows have no pinned C7 r3
# checkpoint). The CPU receipt for the retracted RED is
# artifacts/grm_a1/lt1_margin_first.json. To pin the rule on a GPU row:
#
#   python3 scripts/grm_a1_gpu_contrast.py --rule margin_first \
#       --lease-seconds 560 --out artifacts/grm_a1/gpu_margin_first --resume

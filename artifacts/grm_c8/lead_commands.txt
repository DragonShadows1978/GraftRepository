# EXIT always prints the final summary, including after a RED/preflight failure.
finish() {
    campaign_status=$?
    trap - EXIT
    set +e
    CUDA_VISIBLE_DEVICES='' python scripts/grm_c8_cells.py --summary
    summary_status=$?
    if (( campaign_status != 0 )); then
        exit "$campaign_status"
    fi
    exit "$summary_status"
}
